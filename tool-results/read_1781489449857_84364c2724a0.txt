    52→function animateValue(el, end, dur) {
    53→    if (!dur) dur = 900;
    54→    if (!end) { el.textContent = fmt(end); return; }
    55→    var start = 0;
    56→    var st = null;
    57→    var step = function(ts) {
    58→        if (!st) st = ts;
    59→        var p = Math.min((ts - st) / dur, 1);
    60→        // EaseOutCubic
    61→        var ease = 1 - Math.pow(1 - p, 3);
    62→        el.textContent = fmt(Math.floor(end * ease));
    63→        if (p < 1) requestAnimationFrame(step);
    64→    };
    65→    requestAnimationFrame(step);
    66→}

   321→function parseNgayCap(str) {
   322→    if (!str) return null;
   323→    var s = String(str).trim();
   324→    if (!s) return null;
   325→    // Try DD/MM/YYYY or DD-MM-YYYY first
   326→    var m = s.match(/^(\d{1,2})[\/\-\.](\d{1,2})[\/\-\.](\d{2,4})/);
   327→    if (m) {
   328→        var d = parseInt(m[1], 10), mo = parseInt(m[2], 10), y = parseInt(m[3], 10);
   329→        if (y < 100) y += 2000;
   330→        if (mo >= 1 && mo <= 12 && y > 1900) return { year: y, month: mo };
   331→    }
   332→    // Try YYYY-MM-DD
   333→    m = s.match(/^(\d{4})[\/\-\.](\d{1,2})[\/\-\.](\d{1,2})/);
   334→    if (m) {
   335→        var y2 = parseInt(m[1], 10), mo2 = parseInt(m[2], 10);
   336→        if (mo2 >= 1 && mo2 <= 12 && y2 > 1900) return { year: y2, month: mo2 };
   337→    }
   338→    // Fallback: native Date
   339→    var dt = new Date(s);
   340→    if (!isNaN(dt.getTime())) return { year: dt.getFullYear(), month: dt.getMonth() + 1 };
   341→    return null;
   342→}
   343→function computeGroupStats(members) {
   344→    var total = 0, luotHD = 0, luotHDChuan = 0, tvvMoi = 0;
   345→    var nowY = NOW.getFullYear(), nowM = NOW.getMonth() + 1;
   346→    (members || []).forEach(function(m) {
   347→        total++;
   348→        // Lượt HD: count months (all 12 months) where IP >= 3 (triệu)
   349→        // Lượt HD chuẩn: same but IP >= 12 (triệu)
   350→        var months = m.months || [];
   351→        for (var i = 0; i < 12; i++) {
   352→            var v = +months[i] || 0;
   353→            if (v >= 3) luotHD++;
   354→            if (v >= 12) luotHDChuan++;
   355→        }
   356→        // TVVm (<12t): any member with ngayCap within 12 months from today (count by month, not day)
   357→        var d = parseNgayCap(m.ngayCap);
   358→        if (d) {
   359→            var diff = (nowY - d.year) * 12 + (nowM - d.month);
   360→            if (diff >= 0 && diff <= 12) tvvMoi++;
   361→        }
   362→    });
   363→    return { total: total, luotHD: luotHD, luotHDChuan: luotHDChuan, tvvMoi: tvvMoi };
   364→}

   611→function loadCsv(url) {
   612→    return new Promise(function(resolve, reject) {
   613→        var separator = url.indexOf('?') !== -1 ? '&' : '?';
   614→        Papa.parse(url + separator + 't=' + Date.now(), {
   615→            download: true,
   616→            header: true,
   617→            skipEmptyLines: true,
   618→            transformHeader: function(header) { return normalizeHeaderName(header); },
   619→            complete: function(res) { resolve(normalizeCsvRows(res.data || [])); },
   620→            error: function(err) { reject(err || new Error('Load CSV failed')); }
   621→        });
   622→    });
   623→}

   625→function formatKyLabel(ky) {
   626→    if (!ky) return '--';
   627→    var v = String(ky).trim();
   628→    var monthMatch = v.match(/(\d{2})$/);
   629→    if (/Q[1-4]$/i.test(v)) return v.match(/Q[1-4]$/i)[0].toUpperCase();
   630→    if (/(?:H1|6T)$/i.test(v)) return '6T';
   631→    if (/(?:Y|NAM|NĂM)$/i.test(v)) return 'Năm';
   632→    if (/^\d{4}-\d{2}$/.test(v) && monthMatch) return monthMatch[1];
   633→    if (/^\d{2}$/.test(v)) return v;
   634→    return v.replace(/^\d{4}-/, '');
   635→}

   637→function kySortWeight(ky) {
   638→    var label = formatKyLabel(ky);

   641→    if (label === '6T') return 30;
   642→    if (label === 'Năm') return 31;
   643→    return 99;
   644→}

   825→    var months = [];
   826→    var totalAfyp = 0, totalKh = 0;
   827→    var curM = parseInt(CUR_MONTH, 10);
   828→    for (var m = 1; m <= 12; m++) {
   829→        var mk = CUR_YEAR + '-' + String(m).padStart(2, '0');
   830→        var mRows = DATA.filter(function(x) { return x.ky === mk && x.ten_nhan_vien; });
   831→        var afyp = 0, kh = 0;
   832→        adKeys.forEach(function(adKey) {
   833→            var row = mRows.find(function(x) {
   834→                var ten = normKey(x.ten_nhan_vien);
   835→                var key = normKey(adKey);
   836→                return ten && key && (ten === key || ten.indexOf(key) !== -1 || key.indexOf(ten) !== -1);
   837→            });
   838→            if (row) { afyp += num(row.afyp); kh += num(row.ke_hoach_afyp); }
   839→        });
   840→        months.push({ month: m, label: 'T' + m, afyp: afyp, kh: kh });
   841→        if (m <= curM) { totalAfyp += afyp; totalKh += kh; }
   842→    }
   843→
   844→    var hasData = months.some(function(d) { return d.afyp > 0 || d.kh > 0; });
   845→    if (!hasData) { wrap.style.display = 'none'; return; }
   846→

   848→    var sumEl = $('afyp-chart-summary');
   849→    if (sumEl) {
   850→        var fmtBig = function(v) { return v >= 1e9 ? (v/1e9).toFixed(2) + ' tỷ' : v >= 1e6 ? (v/1e6).toFixed(0) + ' tr' : v >= 1e3 ? (v/1e3).toFixed(0) + 'k' : v.toFixed(0); };
   851→        var avgPct = totalKh > 0 ? (totalAfyp / totalKh * 100) : 0;
   852→        var avgPctClass = avgPct >= 100 ? 'green' : avgPct >= 70 ? 'gold' : 'red';
   853→        sumEl.innerHTML = '<div class="sum-item"><div class="sum-label">Tổng AFYP (T1-T' + curM + ')</div><div class="sum-val">' + fmtBig(totalAfyp) + '</div></div>' +
   854→            '<div class="sum-item"><div class="sum-label">Tổng KH (T1-T' + curM + ')</div><div class="sum-val">' + fmtBig(totalKh) + '</div></div>' +
   855→            '<div class="sum-item"><div class="sum-label">Đạt KH</div><div class="sum-val ' + avgPctClass + '">' + avgPct.toFixed(1) + '%</div></div>';

   871→    if (maxVal / niceMax > 5) niceMax *= 10;
   872→    else if (maxVal / niceMax > 2) niceMax *= 5;
   873→    else if (maxVal / niceMax > 1) niceMax *= 2;
