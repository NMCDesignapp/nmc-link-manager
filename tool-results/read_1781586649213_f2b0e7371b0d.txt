   226→// Smart currency formatting: mobile shows trđ/tỷ, desktop shows full đ
   227→function formatSmartCurrency(amount: number): string {
   228→  const isMobile = typeof window !== 'undefined' && window.innerWidth < 768;
   229→  if (isMobile) {
   230→    if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(2).replace(/\.?0+$/, '')} tỷ`;
   231→    if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(2).replace(/\.?0+$/, '')} trđ`;
   232→    if (amount >= 1_000) return `${(amount / 1_000).toFixed(1).replace(/\.?0+$/, '')} ngàn`;
   233→    return `${amount} đ`;
   234→  }
   235→  return formatCurrency(amount);
   236→}
   237→
   238→// Compact currency for KPI cards - always shows trđ/tỷ/ngàn on ALL screen sizes
   239→// trđ always shows 3 decimal places with Vietnamese comma separator (e.g. 1,350 trđ)
   240→function formatKpiCurrency(amount: number): string {
   241→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(3).replace('.', ',')} tỷ`;
   242→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(3).replace('.', ',')} trđ`;
   243→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(3).replace('.', ',')} ngàn`;
   244→  if (amount === 0) return '0,000 trđ';
   245→  return `${amount} đ`;
   246→}

   248→// Helper: convert any date value to yyyy-mm-dd for <input type="date">
   249→function toInputDate(val: any): string {
   250→  if (!val) return '';
   251→  const s = String(val);
   252→  // Already yyyy-mm-dd
   253→  if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s;
   254→  // ISO string: take first 10 chars
   255→  if (/^\d{4}-\d{2}-\d{2}T/.test(s)) return s.slice(0, 10);
   256→  // dd/mm/yyyy Vietnamese format
   257→  const dmy = s.match(/^(\d{1,2})[./-](\d{1,2})[./-](\d{4})$/);
   258→  if (dmy) return `${dmy[3]}-${dmy[2].padStart(2, '0')}-${dmy[1].padStart(2, '0')}`;
   259→  // Try native parse
   260→  const d = new Date(s);
   261→  if (!isNaN(d.getTime())) {
   262→    const y = d.getFullYear();
   263→    const m = String(d.getMonth() + 1).padStart(2, '0');
   264→    const day = String(d.getDate()).padStart(2, '0');
   265→    return `${y}-${m}-${day}`;
   266→  }
   267→  return '';
   268→}

   270→// Helper: format date value for display (Vietnamese format)
   271→function formatDateDisplay(val: any): string {
   272→  if (!val) return '—';
   273→  const d = new Date(String(val));
   274→  if (isNaN(d.getTime())) return String(val || '—');
   275→  return d.toLocaleDateString('vi-VN');
   276→}

   278→// Helper: safe format date without timezone offset issues
   279→function safeFormatDate(val: any): string {
   280→  if (!val) return '';
   281→  const s = String(val);
   282→  // If it's a date-only or ISO string like "2026-01-01" or "2026-01-01T00:00:00.000Z", parse manually
   283→  const ymd = s.match(/^(\d{4})-(\d{2})-(\d{2})/);
   284→  if (ymd) return `${ymd[3]}/${ymd[2]}/${ymd[1]}`;
   285→  // Otherwise use locale
   286→  const d = new Date(s);
   287→  if (!isNaN(d.getTime())) return d.toLocaleDateString('vi-VN');
   288→  return s;
   289→}

   291→// ==================== EDITABLE CELL ====================
   292→function EditableCell({ value, onSave, type = 'text', className = '' }: {
   293→  value: string | number; onSave: (val: any) => void; type?: 'text' | 'number' | 'date'; className?: string;
   294→}) {
   295→  const [editing, setEditing] = useState(false);
   296→  const [editVal, setEditVal] = useState('');
   297→  const inputRef = useRef<HTMLInputElement>(null);
   298→
   299→  // Sync editVal with value, converting dates properly
   300→  useEffect(() => {
   301→    if (type === 'date') {
   302→      setEditVal(toInputDate(value));
   303→    } else {
   304→      setEditVal(String(value ?? ''));
   305→    }
   306→  }, [value, type]);
   307→  useEffect(() => { if (editing && inputRef.current) inputRef.current.focus(); }, [editing]);
   308→
   309→  const handleSave = useCallback(() => {
   310→    const finalVal = type === 'number' ? (parseFloat(editVal) || 0) : editVal;
   311→    if (finalVal !== value) onSave(finalVal);
   312→    setEditing(false);
   313→  }, [editVal, value, type, onSave]);
   314→
   315→  if (editing) {
   316→    return (
   317→      <input
   318→        ref={inputRef}
   319→        type={type === 'number' ? 'number' : type === 'date' ? 'date' : 'text'}
   320→        value={editVal}
   321→        onChange={(e) => setEditVal(e.target.value)}
   322→        onBlur={handleSave}
   323→        onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') setEditing(false); }}
   324→        className="w-full h-full px-1 py-0.5 text-xs bg-white text-gray-800 border-2 border-emerald-400 outline-none"
   325→      />
   326→    );
   327→  }
   328→
   329→  return (
   330→    <div
   331→      className={`cursor-pointer hover:bg-emerald-50 hover:outline hover:outline-1 hover:outline-emerald-400 px-1 py-0.5 min-h-[22px] text-gray-800 ${className}`}
   332→      onDoubleClick={() => setEditing(true)}
   333→      title="Nháy đúp để sửa"
   334→    >
   335→      {type === 'number' && typeof value === 'number'
   336→        ? formatNumber(value)
   337→        : type === 'date'
   338→          ? formatDateDisplay(value)
   339→          : String(value || '—')}
   340→    </div>
   341→  );
   342→}

   345→function SettingsPopover({ sectionKey, sectionLabel, onlineSettings, saveSetting }: { sectionKey: string; sectionLabel: string; onlineSettings: Record<string, string>; saveSetting: (key: string, value: string) => Promise<void> }) {
   346→  const [link, setLink] = useState('');
   347→  const [syncOn, setSyncOn] = useState(true);
   348→  const [open, setOpen] = useState(false);
   349→
   350→  useEffect(() => {
   351→    const savedLink = onlineSettings[`nmc-link-${sectionKey}`] || '';
   352→    const savedSync = onlineSettings[`nmc-sync-${sectionKey}`];
   353→    setLink(savedLink);
   354→    if (savedSync !== undefined && savedSync !== '') setSyncOn(savedSync === 'true');
   355→  }, [sectionKey, open, onlineSettings]);
   356→
   357→  const handleSave = useCallback(() => {
   358→    saveSetting(`nmc-link-${sectionKey}`, link);
   359→    saveSetting(`nmc-sync-${sectionKey}`, String(syncOn));
   360→    toast({ title: 'Đã lưu cài đặt', description: `${sectionLabel}: ${link ? 'Đã thiết lập link' : 'Chưa có link'} • Đồng bộ: ${syncOn ? 'BẬT' : 'TẮT'}` });
   361→    setOpen(false);
   362→  }, [link, syncOn, sectionKey, sectionLabel, saveSetting]);
   363→
   364→  const hasLink = !!(onlineSettings[`nmc-link-${sectionKey}`]);
   365→
   366→  return (
   367→    <Popover open={open} onOpenChange={setOpen}>
   368→      <PopoverTrigger asChild>
   369→        <Button
   370→          variant="ghost"
   371→          className={`h-8 px-2 text-xs ${hasLink ? 'text-emerald-300 hover:text-emerald-200' : 'text-gray-400 hover:text-gray-200'} hover:bg-emerald-500/10`}
   372→          title="Cài đặt"
   373→        >
   374→          <Settings className="w-3.5 h-3.5" />
   375→          {hasLink && <Link2 className="w-3 h-3 ml-1 text-emerald-400" />}
   376→        </Button>
   377→      </PopoverTrigger>
   378→      <PopoverContent className="bg-[#0e0e18]/95 backdrop-blur-xl border-emerald-500/30 w-80 p-3" align="end" sideOffset={4}>

   656→function evaluateFormula(formula: string, cells: CellMap): string {
   657→  if (!formula.startsWith('=')) return formula;
   658→  const expr = formula.slice(1).trim().toUpperCase();
   659→
   660→  // SUM function
   661→  const sumMatch = expr.match(/^SUM\(([A-Z]\d+):([A-Z]\d+)\)$/);
   662→  if (sumMatch) {
   663→    const startCol = sumMatch[1][0], startRow = parseInt(sumMatch[1].slice(1));
   664→    const endCol = sumMatch[2][0], endRow = parseInt(sumMatch[2].slice(1));
   665→    let total = 0;
   666→    for (let r = startRow; r <= endRow; r++) {
   667→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   668→        const val = parseFloat(cells[String.fromCharCode(c) + r] || '0');
   669→        if (!isNaN(val)) total += val;
   670→      }
   671→    }
   672→    return String(total);
   673→  }
   674→
   675→  // AVERAGE function
   676→  const avgMatch = expr.match(/^AVERAGE\(([A-Z]\d+):([A-Z]\d+)\)$/);
   677→  if (avgMatch) {
   678→    const startCol = avgMatch[1][0], startRow = parseInt(avgMatch[1].slice(1));
   679→    const endCol = avgMatch[2][0], endRow = parseInt(avgMatch[2].slice(1));
   680→    let total = 0, count = 0;
   681→    for (let r = startRow; r <= endRow; r++) {
   682→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   683→        const val = parseFloat(cells[String.fromCharCode(c) + r] || '0');
   684→        if (!isNaN(val) && cells[String.fromCharCode(c) + r]) { total += val; count++; }
   685→      }
   686→    }
   687→    return count > 0 ? String(Math.round((total / count) * 100) / 100) : '0';
   688→  }
   689→
   690→  // COUNT function
   691→  const countMatch = expr.match(/^COUNT\(([A-Z]\d+):([A-Z]\d+)\)$/);
   692→  if (countMatch) {
   693→    const startCol = countMatch[1][0], startRow = parseInt(countMatch[1].slice(1));
   694→    const endCol = countMatch[2][0], endRow = parseInt(countMatch[2].slice(1));
   695→    let count = 0;
   696→    for (let r = startRow; r <= endRow; r++) {
   697→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   698→        if (cells[String.fromCharCode(c) + r]) count++;
   699→      }
   700→    }
   701→    return String(count);
   702→  }
   703→
   704→  // MAX function
   705→  const maxMatch = expr.match(/^MAX\(([A-Z]\d+):([A-Z]\d+)\)$/);
   706→  if (maxMatch) {
   707→    const startCol = maxMatch[1][0], startRow = parseInt(maxMatch[1].slice(1));
   708→    const endCol = maxMatch[2][0], endRow = parseInt(maxMatch[2].slice(1));
   709→    let max = -Infinity;
   710→    for (let r = startRow; r <= endRow; r++) {
   711→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   712→        const val = parseFloat(cells[String.fromCharCode(c) + r] || '0');
   713→        if (!isNaN(val) && cells[String.fromCharCode(c) + r]) max = Math.max(max, val);
   714→      }
   715→    }
   716→    return max === -Infinity ? '0' : String(max);
   717→  }
   718→
   719→  // MIN function
   720→  const minMatch = expr.match(/^MIN\(([A-Z]\d+):([A-Z]\d+)\)$/);
   721→  if (minMatch) {
   722→    const startCol = minMatch[1][0], startRow = parseInt(minMatch[1].slice(1));
   723→    const endCol = minMatch[2][0], endRow = parseInt(minMatch[2].slice(1));
   724→    let min = Infinity;
   725→    for (let r = startRow; r <= endRow; r++) {
   726→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   727→        const val = parseFloat(cells[String.fromCharCode(c) + r] || '0');
   728→        if (!isNaN(val) && cells[String.fromCharCode(c) + r]) min = Math.min(min, val);
   729→      }
   730→    }
   731→    return min === Infinity ? '0' : String(min);
   732→  }
   733→
   734→  // Cell reference math: =A1+B1, =A1*2, =A1+B1*C1
   735→  try {
   736→    const resolved = expr.replace(/([A-Z])(\d+)/g, (match) => {
   737→      const v = parseFloat(cells[match] || '0');
   738→      return isNaN(v) ? '0' : String(v);
   739→    });
   740→    // Safe eval for simple math
   741→    const result = Function('"use strict"; return (' + resolved + ')')();
   742→    if (typeof result === 'number' && isFinite(result)) return String(Math.round(result * 100) / 100);
   743→    return formula;
   744→  } catch {
   745→    return formula;
   746→  }
   747→}

   758→  const initializedRef = useRef(false);
   759→
   760→  // Load cells and merges from parent's onlineSettings (no extra API call)
   761→  useEffect(() => {
   762→    if (initializedRef.current) return;
   763→    const cellsData = onlineSettings['nmc-spreadsheet-cells'];
   764→    const mergesData = onlineSettings['nmc-spreadsheet-merges'];
   765→    if (cellsData) { try { setCells(JSON.parse(cellsData)); } catch {} }
   766→    if (mergesData) { try { setMerges(JSON.parse(mergesData)); } catch {} }
   767→    if (Object.keys(onlineSettings).length > 0) initializedRef.current = true;
   768→  }, [onlineSettings]);
   769→

   771→  const saveCellsTimeout = useRef<NodeJS.Timeout | null>(null);
   772→  useEffect(() => {
   773→    if (Object.keys(cells).length === 0 || !initializedRef.current) return;
   774→    if (saveCellsTimeout.current) clearTimeout(saveCellsTimeout.current);

   782→  useEffect(() => {

   784→    if (saveMergesTimeout.current) clearTimeout(saveMergesTimeout.current);
   785→    saveMergesTimeout.current = setTimeout(() => {
   786→      saveSetting('nmc-spreadsheet-merges', JSON.stringify(merges));
   787→    }, 1000);
   788→  }, [merges, saveSetting]);
   789→
   790→  useEffect(() => {
   791→    if (editingCell && inputRef.current) inputRef.current.focus();
   792→  }, [editingCell]);
   793→
   794→  // Parse cellId to {col, row} (0-indexed)
   795→  const parseCellId = useCallback((cellId: string) => {
   796→    const col = cellId.charCodeAt(0) - 65;
   797→    const row = parseInt(cellId.slice(1)) - 1;
   798→    return { col, row };
   799→  }, []);
   800→
   801→  // Check if a cell is the top-left of a merge
   802→  const getCellMerge = useCallback((cellId: string): MergeRange | null => {
   803→    const { col, row } = parseCellId(cellId);
   804→    return merges.find(m => m.startRow === row && m.startCol === col) || null;
   805→  }, [merges, parseCellId]);
   806→
   807→  // Check if a cell is hidden by a merge
   808→  const isCellHidden = useCallback((cellId: string): boolean => {
   809→    const { col, row } = parseCellId(cellId);
   810→    return merges.some(m =>
   811→      row >= m.startRow && row <= m.endRow &&
   812→      col >= m.startCol && col <= m.endCol &&
   813→      !(row === m.startRow && col === m.startCol)
   814→    );
   815→  }, [merges, parseCellId]);

   834→    const minCol = Math.min(sc, ec), maxCol = Math.max(sc, ec);
   835→    const minRow = Math.min(sr, er), maxRow = Math.max(sr, er);
   836→    return col >= minCol && col <= maxCol && row >= minRow && row <= maxRow;

   849→      startCol: Math.min(sc, ec),

  1055→    </div>
