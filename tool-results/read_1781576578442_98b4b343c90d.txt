   451→   451→function num(v: unknown): number {
   452→   452→  if (v === null || v === undefined) return 0;
   453→   453→  if (typeof v === 'number') return isFinite(v) ? v : 0;
   454→   454→  const s = String(v).trim().replace(/\s/g, '').replace(/%/g, '');
   455→   455→  if (!s) return 0;
   456→   456→  const n = Number(s.replace(/[^0-9.\-]/g, ''));
   457→   457→  return isFinite(n) ? n : 0;
   458→   458→}
