   227→// Smart currency formatting: mobile shows trđ/tỷ, desktop shows full đ
   228→function formatSmartCurrency(amount: number): string {
   229→  const isMobile = typeof window !== 'undefined' && window.innerWidth < 768;
   230→  if (isMobile) {
   231→    if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(2).replace(/\.?0+$/, '')} tỷ`;
   232→    if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(2).replace(/\.?0+$/, '')} trđ`;
   233→    if (amount >= 1_000) return `${(amount / 1_000).toFixed(1).replace(/\.?0+$/, '')} ngàn`;
   234→    return `${amount} đ`;
   235→  }
   236→  return formatCurrency(amount);
   237→}
   238→
   239→// Compact currency for KPI cards - always shows trđ/tỷ/ngàn on ALL screen sizes
   240→// trđ always shows 3 decimal places with Vietnamese comma separator (e.g. 1,350 trđ)
   241→function formatKpiCurrency(amount: number): string {
   242→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(3).replace('.', ',')} tỷ`;
   243→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(3).replace('.', ',')} trđ`;
   244→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(3).replace('.', ',')} ngàn`;
   245→  if (amount === 0) return '0,000 trđ';
   246→  return `${amount} đ`;
   247→}

   249→// Helper: convert any date value to yyyy-mm-dd for <input type="date">
   250→function toInputDate(val: any): string {
   251→  if (!val) return '';
   252→  const s = String(val);
   253→  // Already yyyy-mm-dd
   254→  if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s;
   255→  // ISO string: take first 10 chars
   256→  if (/^\d{4}-\d{2}-\d{2}T/.test(s)) return s.slice(0, 10);
   257→  // dd/mm/yyyy Vietnamese format
   258→  const dmy = s.match(/^(\d{1,2})[./-](\d{1,2})[./-](\d{4})$/);
   259→  if (dmy) return `${dmy[3]}-${dmy[2].padStart(2, '0')}-${dmy[1].padStart(2, '0')}`;
   260→  // Try native parse
   261→  const d = new Date(s);
   262→  if (!isNaN(d.getTime())) {
   263→    const y = d.getFullYear();
   264→    const m = String(d.getMonth() + 1).padStart(2, '0');
   265→    const day = String(d.getDate()).padStart(2, '0');
   266→    return `${y}-${m}-${day}`;
   267→  }
   268→  return '';
   269→}

   271→// Helper: format date value for display (Vietnamese format)
   272→function formatDateDisplay(val: any): string {
   273→  if (!val) return '—';
   274→  const d = new Date(String(val));
   275→  if (isNaN(d.getTime())) return String(val || '—');
   276→  return d.toLocaleDateString('vi-VN');
   277→}

   279→// Helper: safe format date without timezone offset issues
   280→function safeFormatDate(val: any): string {
   281→  if (!val) return '';
   282→  const s = String(val);
   283→  // If it's a date-only or ISO string like "2026-01-01" or "2026-01-01T00:00:00.000Z", parse manually
   284→  const ymd = s.match(/^(\d{4})-(\d{2})-(\d{2})/);
   285→  if (ymd) return `${ymd[3]}/${ymd[2]}/${ymd[1]}`;
   286→  // Otherwise use locale
   287→  const d = new Date(s);
   288→  if (!isNaN(d.getTime())) return d.toLocaleDateString('vi-VN');
   289→  return s;
   290→}

   292→// ==================== EDITABLE CELL ====================
   293→function EditableCell({ value, onSave, type = 'text', className = '' }: {
   294→  value: string | number; onSave: (val: any) => void; type?: 'text' | 'number' | 'date'; className?: string;
   295→}) {
   296→  const [editing, setEditing] = useState(false);
   297→  const [editVal, setEditVal] = useState('');
   298→  const inputRef = useRef<HTMLInputElement>(null);
   299→
   300→  // Sync editVal with value, converting dates properly
   301→  useEffect(() => {
   302→    if (type === 'date') {
   303→      setEditVal(toInputDate(value));
   304→    } else {
   305→      setEditVal(String(value ?? ''));
   306→    }
   307→  }, [value, type]);
   308→  useEffect(() => { if (editing && inputRef.current) inputRef.current.focus(); }, [editing]);
   309→
   310→  const handleSave = useCallback(() => {
   311→    const finalVal = type === 'number' ? (parseFloat(editVal) || 0) : editVal;
   312→    if (finalVal !== value) onSave(finalVal);
   313→    setEditing(false);
   314→  }, [editVal, value, type, onSave]);
   315→
   316→  if (editing) {
   317→    return (
   318→      <input
   319→        ref={inputRef}
   320→        type={type === 'number' ? 'number' : type === 'date' ? 'date' : 'text'}
   321→        value={editVal}
   322→        onChange={(e) => setEditVal(e.target.value)}
   323→        onBlur={handleSave}
   324→        onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') setEditing(false); }}
   325→        className="w-full h-full px-1 py-0.5 text-xs bg-white text-gray-800 border-2 border-emerald-400 outline-none"
   326→      />
   327→    );
   328→  }
   329→
   330→  return (
   331→    <div
   332→      className={`cursor-pointer hover:bg-emerald-50 hover:outline hover:outline-1 hover:outline-emerald-400 px-1 py-0.5 min-h-[22px] text-gray-800 ${className}`}
   333→      onDoubleClick={() => setEditing(true)}
   334→      title="Nháy đúp để sửa"
   335→    >
   336→      {type === 'number' && typeof value === 'number'
   337→        ? formatNumber(value)
   338→        : type === 'date'
   339→          ? formatDateDisplay(value)
   340→          : String(value || '—')}
   341→    </div>
   342→  );
   343→}

   346→function SettingsPopover({ sectionKey, sectionLabel, onlineSettings, saveSetting }: { sectionKey: string; sectionLabel: string; onlineSettings: Record<string, string>; saveSetting: (key: string, value: string) => Promise<void> }) {
   347→  const [link, setLink] = useState('');
   348→  const [syncOn, setSyncOn] = useState(true);
   349→  const [open, setOpen] = useState(false);
   350→
   351→  useEffect(() => {
   352→    const savedLink = onlineSettings[`nmc-link-${sectionKey}`] || '';
   353→    const savedSync = onlineSettings[`nmc-sync-${sectionKey}`];
   354→    setLink(savedLink);
   355→    if (savedSync !== undefined && savedSync !== '') setSyncOn(savedSync === 'true');
   356→  }, [sectionKey, open, onlineSettings]);
   357→
   358→  const handleSave = useCallback(() => {
   359→    saveSetting(`nmc-link-${sectionKey}`, link);
   360→    saveSetting(`nmc-sync-${sectionKey}`, String(syncOn));
   361→    toast({ title: 'Đã lưu cài đặt', description: `${sectionLabel}: ${link ? 'Đã thiết lập link' : 'Chưa có link'} • Đồng bộ: ${syncOn ? 'BẬT' : 'TẮT'}` });
   362→    setOpen(false);
   363→  }, [link, syncOn, sectionKey, sectionLabel, saveSetting]);
   364→
   365→  const hasLink = !!(onlineSettings[`nmc-link-${sectionKey}`]);
   366→
   367→  return (
   368→    <Popover open={open} onOpenChange={setOpen}>
   369→      <PopoverTrigger asChild>
   370→        <Button
   371→          variant="ghost"
   372→          className={`h-8 px-2 text-xs ${hasLink ? 'text-emerald-300 hover:text-emerald-200' : 'text-gray-400 hover:text-gray-200'} hover:bg-emerald-500/10`}
   373→          title="Cài đặt"
   374→        >
   375→          <Settings className="w-3.5 h-3.5" />
   376→          {hasLink && <Link2 className="w-3 h-3 ml-1 text-emerald-400" />}
   377→        </Button>
   378→      </PopoverTrigger>
   379→      <PopoverContent className="bg-[#0e0e18]/95 backdrop-blur-xl border-emerald-500/30 w-80 p-3" align="end" sideOffset={4}>

   657→function evaluateFormula(formula: string, cells: CellMap): string {
   658→  if (!formula.startsWith('=')) return formula;
   659→  const expr = formula.slice(1).trim().toUpperCase();
   660→
   661→  // SUM function
   662→  const sumMatch = expr.match(/^SUM\(([A-Z]\d+):([A-Z]\d+)\)$/);
   663→  if (sumMatch) {
   664→    const startCol = sumMatch[1][0], startRow = parseInt(sumMatch[1].slice(1));
   665→    const endCol = sumMatch[2][0], endRow = parseInt(sumMatch[2].slice(1));
   666→    let total = 0;
   667→    for (let r = startRow; r <= endRow; r++) {
   668→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   669→        const val = parseFloat(cells[String.fromCharCode(c) + r] || '0');
   670→        if (!isNaN(val)) total += val;
   671→      }
   672→    }
   673→    return String(total);
   674→  }
   675→
   676→  // AVERAGE function
   677→  const avgMatch = expr.match(/^AVERAGE\(([A-Z]\d+):([A-Z]\d+)\)$/);
   678→  if (avgMatch) {
   679→    const startCol = avgMatch[1][0], startRow = parseInt(avgMatch[1].slice(1));
   680→    const endCol = avgMatch[2][0], endRow = parseInt(avgMatch[2].slice(1));
   681→    let total = 0, count = 0;
   682→    for (let r = startRow; r <= endRow; r++) {
   683→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   684→        const val = parseFloat(cells[String.fromCharCode(c) + r] || '0');
   685→        if (!isNaN(val) && cells[String.fromCharCode(c) + r]) { total += val; count++; }
   686→      }
   687→    }
   688→    return count > 0 ? String(Math.round((total / count) * 100) / 100) : '0';
   689→  }
   690→
   691→  // COUNT function
   692→  const countMatch = expr.match(/^COUNT\(([A-Z]\d+):([A-Z]\d+)\)$/);
   693→  if (countMatch) {
   694→    const startCol = countMatch[1][0], startRow = parseInt(countMatch[1].slice(1));
   695→    const endCol = countMatch[2][0], endRow = parseInt(countMatch[2].slice(1));
   696→    let count = 0;
   697→    for (let r = startRow; r <= endRow; r++) {
   698→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   699→        if (cells[String.fromCharCode(c) + r]) count++;
   700→      }
   701→    }
   702→    return String(count);
   703→  }
   704→
   705→  // MAX function
   706→  const maxMatch = expr.match(/^MAX\(([A-Z]\d+):([A-Z]\d+)\)$/);
   707→  if (maxMatch) {
   708→    const startCol = maxMatch[1][0], startRow = parseInt(maxMatch[1].slice(1));
   709→    const endCol = maxMatch[2][0], endRow = parseInt(maxMatch[2].slice(1));
   710→    let max = -Infinity;
   711→    for (let r = startRow; r <= endRow; r++) {
   712→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   713→        const val = parseFloat(cells[String.fromCharCode(c) + r] || '0');
   714→        if (!isNaN(val) && cells[String.fromCharCode(c) + r]) max = Math.max(max, val);
   715→      }
   716→    }
   717→    return max === -Infinity ? '0' : String(max);
   718→  }
   719→
   720→  // MIN function
   721→  const minMatch = expr.match(/^MIN\(([A-Z]\d+):([A-Z]\d+)\)$/);
   722→  if (minMatch) {
   723→    const startCol = minMatch[1][0], startRow = parseInt(minMatch[1].slice(1));
   724→    const endCol = minMatch[2][0], endRow = parseInt(minMatch[2].slice(1));
   725→    let min = Infinity;
   726→    for (let r = startRow; r <= endRow; r++) {
   727→      for (let c = startCol.charCodeAt(0); c <= endCol.charCodeAt(0); c++) {
   728→        const val = parseFloat(cells[String.fromCharCode(c) + r] || '0');
   729→        if (!isNaN(val) && cells[String.fromCharCode(c) + r]) min = Math.min(min, val);
   730→      }
   731→    }
   732→    return min === Infinity ? '0' : String(min);
   733→  }
   734→
   735→  // Cell reference math: =A1+B1, =A1*2, =A1+B1*C1
   736→  try {
   737→    const resolved = expr.replace(/([A-Z])(\d+)/g, (match) => {
   738→      const v = parseFloat(cells[match] || '0');
   739→      return isNaN(v) ? '0' : String(v);
   740→    });
   741→    // Safe eval for simple math
   742→    const result = Function('"use strict"; return (' + resolved + ')')();
   743→    if (typeof result === 'number' && isFinite(result)) return String(Math.round(result * 100) / 100);
   744→    return formula;
   745→  } catch {
   746→    return formula;
   747→  }
   748→}

   759→  const initializedRef = useRef(false);
   760→
   761→  // Load cells and merges from parent's onlineSettings (no extra API call)
   762→  useEffect(() => {
   763→    if (initializedRef.current) return;
   764→    const cellsData = onlineSettings['nmc-spreadsheet-cells'];
   765→    const mergesData = onlineSettings['nmc-spreadsheet-merges'];
   766→    if (cellsData) { try { setCells(JSON.parse(cellsData)); } catch {} }
   767→    if (mergesData) { try { setMerges(JSON.parse(mergesData)); } catch {} }
   768→    if (Object.keys(onlineSettings).length > 0) initializedRef.current = true;
   769→  }, [onlineSettings]);
   770→

   772→  const saveCellsTimeout = useRef<NodeJS.Timeout | null>(null);
   773→  useEffect(() => {
   774→    if (Object.keys(cells).length === 0 || !initializedRef.current) return;
   775→    if (saveCellsTimeout.current) clearTimeout(saveCellsTimeout.current);
   776→    saveCellsTimeout.current = setTimeout(() => {
   777→      saveSetting('nmc-spreadsheet-cells', JSON.stringify(cells));

   779→  }, [cells, saveSetting]);
   780→
   781→  // Save merges to online API (debounced)
   782→  const saveMergesTimeout = useRef<NodeJS.Timeout | null>(null);
   783→  useEffect(() => {
   784→    if (merges.length === 0 || !initializedRef.current) return;
   785→    if (saveMergesTimeout.current) clearTimeout(saveMergesTimeout.current);
   786→    saveMergesTimeout.current = setTimeout(() => {
   787→      saveSetting('nmc-spreadsheet-merges', JSON.stringify(merges));
   788→    }, 1000);
   789→  }, [merges, saveSetting]);
   790→
   791→  useEffect(() => {
   792→    if (editingCell && inputRef.current) inputRef.current.focus();
   793→  }, [editingCell]);
   794→
   795→  // Parse cellId to {col, row} (0-indexed)
   796→  const parseCellId = useCallback((cellId: string) => {
   797→    const col = cellId.charCodeAt(0) - 65;
   798→    const row = parseInt(cellId.slice(1)) - 1;
   799→    return { col, row };
   800→  }, []);
   801→
   802→  // Check if a cell is the top-left of a merge
   803→  const getCellMerge = useCallback((cellId: string): MergeRange | null => {
   804→    const { col, row } = parseCellId(cellId);
   805→    return merges.find(m => m.startRow === row && m.startCol === col) || null;
   806→  }, [merges, parseCellId]);
   807→
   808→  // Check if a cell is hidden by a merge
   809→  const isCellHidden = useCallback((cellId: string): boolean => {
   810→    const { col, row } = parseCellId(cellId);
   811→    return merges.some(m =>
   812→      row >= m.startRow && row <= m.endRow &&
   813→      col >= m.startCol && col <= m.endCol &&
   814→      !(row === m.startRow && col === m.startCol)
   815→    );
   816→  }, [merges, parseCellId]);

   823→      rowSpan: merge.endRow - merge.startRow + 1,

   835→    const minCol = Math.min(sc, ec), maxCol = Math.max(sc, ec);
   836→    const minRow = Math.min(sr, er), maxRow = Math.max(sr, er);
   837→    return col >= minCol && col <= maxCol && row >= minRow && row <= maxRow;
   838→  }, [selectionStart, selectionEnd, parseCellId]);

   850→      startCol: Math.min(sc, ec),

   904→      }

  1055→      <div className="text-gray-400 text-[10px] mt-1">Nháy đúp để sửa ô • Dùng = để nhập công thức • Click+Shift để chọn vùng • Gộp/Tách ô • Dữ liệu lưu trong trình duyệt</div>
  1056→    </div>

  1655→  // CRITICAL FIX: XLSX cellDates:true creates Date in LOCAL timezone (UTC+7),
  1656→  // but previous code used getUTC* methods → dates shifted back 1 day

  1659→  const parseDateValue = useCallback((val: any): string | null => {
  1660→    if (!val || val === '' || val === '—') return null;
  1661→    // Already a Date object - XLSX with cellDates:true creates LOCAL timezone Date objects
  1662→    // MUST use getFullYear/getMonth/getDate (NOT getUTC*) because XLSX creates dates in local tz
  1663→    // Example: 01/05/2026 in Excel → new Date(2026,4,1) → local midnight → getUTCDate()=30 (wrong!)
  1664→    if (val instanceof Date) {
  1665→      const y = val.getFullYear();
  1666→      const m = String(val.getMonth() + 1).padStart(2, '0');
  1667→      const d = String(val.getDate()).padStart(2, '0');
  1668→      // Validate: reasonable year range
  1669→      if (y < 1900 || y > 2100) return null;
  1670→      return `${y}-${m}-${d}`;
  1671→    }
  1672→    // String date formats
  1673→    if (typeof val === 'string') {
  1674→      // yyyy-mm-dd
  1675→      if (/^\d{4}-\d{2}-\d{2}/.test(val)) return val.slice(0, 10);
  1676→      // dd/mm/yyyy (Vietnamese format)
  1677→      const dmy = val.match(/^(\d{1,2})[./-](\d{1,2})[./-](\d{4})$/);
  1678→      if (dmy) return `${dmy[3]}-${dmy[2].padStart(2, '0')}-${dmy[1].padStart(2, '0')}`;
  1679→      // Try native parse - use local methods since non-ISO strings create local tz Dates
  1680→      const d = new Date(val);
  1681→      if (!isNaN(d.getTime())) {
  1682→        const y = d.getFullYear();
  1683→        const m = String(d.getMonth() + 1).padStart(2, '0');
  1684→        const day = String(d.getDate()).padStart(2, '0');
  1685→        if (y < 1900 || y > 2100) return null;
  1686→        return `${y}-${m}-${day}`;
  1687→      }
  1688→    }
  1689→    // Excel serial number (days since 1899-12-30)
  1690→    // Serial numbers are timezone-neutral, use UTC conversion
  1691→    if (typeof val === 'number' && val > 1000) {
  1692→      const d = new Date((val - 25569) * 86400 * 1000);
  1693→      if (!isNaN(d.getTime())) {
  1694→        const y = d.getUTCFullYear();
  1695→        const m = String(d.getUTCMonth() + 1).padStart(2, '0');
  1696→        const day = String(d.getUTCDate()).padStart(2, '0');
  1697→        if (y < 1900 || y > 2100) return null;
  1698→        return `${y}-${m}-${day}`;
  1699→      }
  1700→    }
  1701→    return null;
