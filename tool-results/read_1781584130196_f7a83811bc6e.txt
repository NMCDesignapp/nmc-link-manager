   120→function isTVVm(startDate: string | null, maxMonths: number = 12): boolean {
   121→  if (!startDate) return false;
   122→  const start = new Date(startDate);
   123→  const now = new Date();
   124→  const diffMonths = (now.getFullYear() - start.getFullYear()) * 12 + (now.getMonth() - start.getMonth());
   125→  return diffMonths <= maxMonths;
   126→}

   128→// TVV90: TVV có thời gian làm việc không quá N tháng
   129→function isTVV90Agent(contracts: Contract[], agentCode: string, maxMonths: number = 3, _minIP?: number): boolean {
   130→  const agentContract = contracts.find(c => c.agentCode === agentCode);
   131→  if (!agentContract) return false;
   132→  const startDate = agentContract.ngayBatDauLamViec || agentContract.startDate;
   133→  if (!startDate) return false;
   134→  const start = new Date(startDate);
   135→  const now = new Date();
   136→  const diffMonths = (now.getFullYear() - start.getFullYear()) * 12 + (now.getMonth() - start.getMonth());
   137→  return diffMonths <= maxMonths;
   138→}

   140→// Helper: calculate lượt for a group of contracts based on tinhLuot3tr
   141→// Đếm SỐ DÒNG hợp đồng có tinhLuot3tr >= threshold (không phải unique TVV)
   142→function calculateLuot(contracts: Contract[], luotThreshold: number, conditionType: ConditionType, tvv90MaxMonths?: number, tvv90MinIP?: number): number {
   143→  let count = 0;
   144→  for (const c of contracts) {
   145→    // Apply TVVm filter if condition is tvvm mode
   146→    if (isTVVmMode(conditionType)) {
   147→      if (!isTVVm(c.ngayBatDauLamViec || c.startDate)) continue;
   148→    }
   149→    // Apply TVV90 filter if condition is tvv90 mode
   150→    if (conditionType === 'activity_round_tvv90') {
   151→      if (!isTVV90Agent(contracts, c.agentCode, tvv90MaxMonths, tvv90MinIP)) continue;
   152→    }
   153→    // Đếm số dòng hợp đồng có tinhLuot3tr >= threshold
   154→    if (c.tinhLuot3tr >= luotThreshold) {
   155→      count++;
   156→    }
   157→  }
   158→  return count;
   159→}

   174→function formatBonus(tier: BonusTier, fyp?: number, rounds?: number): string {
   175→  if (tier.bonusType === 'gift' && tier.bonusText) return tier.bonusText;
   176→  if (tier.bonusType === 'percent' && tier.bonusPercent > 0) {
   177→    const calculated = fyp ? tier.bonusPercent / 100 * fyp : 0;
   178→    return `${tier.bonusPercent}% IP${fyp ? ` = ${formatCurrency(calculated)}` : ''}`;
   179→  }
   180→  if (tier.bonusType === 'percent_fyc' && tier.bonusPercent > 0) {
   181→    const fyc = fyp ? fyp * 0.25 : 0;
   182→    const calculated = fyc ? tier.bonusPercent / 100 * fyc : 0;
   183→    return `${tier.bonusPercent}% FYC${fyp ? ` = ${formatCurrency(calculated)}` : ''}`;
   184→  }
   185→  if (tier.bonusType === 'money_per_round') {
   186→    const calculated = rounds ? tier.bonusAmount * rounds : 0;
   187→    return rounds ? `${formatCurrency(tier.bonusAmount)}/lượt × ${rounds} = ${formatCurrency(calculated)}` : `${formatCurrency(tier.bonusAmount)}/lượt`;
   188→  }
   189→  if (tier.bonusType === 'money_per_tvv') {
   190→    const tvvCount = rounds || 0; // rounds param doubles as tvvCount for tvv_pass_count
   191→    const calculated = tvvCount ? tier.bonusAmount * tvvCount : 0;
   192→    return tvvCount ? `${formatCurrency(tier.bonusAmount)}/TVV × ${tvvCount} = ${formatCurrency(calculated)}` : `${formatCurrency(tier.bonusAmount)}/TVV`;
   193→  }
   194→  return formatCurrency(tier.bonusAmount);
   195→}

   205→function formatBonusAmount(tier: BonusTier, fyp?: number, rounds?: number): string {
   206→  if (tier.bonusType === 'gift' && tier.bonusText) return tier.bonusText;
   207→  const amount = computeBonusFromTier(tier, fyp || 0, rounds);
   208→  return formatCurrency(amount);

   216→function computeBonusFromTier(tier: BonusTier, fyp: number, rounds?: number): number {
   217→  if (tier.bonusType === 'percent') return tier.bonusPercent / 100 * fyp;
   218→  if (tier.bonusType === 'percent_fyc') return tier.bonusPercent / 100 * (fyp * 0.25);
   219→  if (tier.bonusType === 'money_per_round') return tier.bonusAmount * (rounds || 0);
   220→  if (tier.bonusType === 'money_per_tvv') return tier.bonusAmount * (rounds || 0); // rounds doubles as tvvCount
   221→  return tier.bonusAmount;
   222→}
