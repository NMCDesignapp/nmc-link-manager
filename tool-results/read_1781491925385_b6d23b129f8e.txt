    59→// Creative N•M•C text with neon effects - horizontal layout with flowing chevron arrows
    60→function NMCLogo({ color = '#00ff88' }: { color?: string }) {
    61→  const parts = ['N', '•', 'M', '•', 'C']
    62→
    63→  // Flowing chevron arrow stream - spans from edge toward center text
    64→  // Uses SVG with animated sweep mask for ultra-smooth flowing light effect
    65→  const ArrowStream = ({ direction }: { direction: 'left' | 'right' }) => {
    66→    const isLeft = direction === 'left'
    67→    const count = 18
    68→    const spacing = 14
    69→    const svgW = count * spacing
    70→    const svgH = 22
    71→
    72→    return (
    73→      <div className="flex-1 min-w-0 overflow-hidden" style={{ height: '24px' }}>
    74→        <svg
    75→          width="100%"
    76→          height="100%"
    77→          viewBox={`0 0 ${svgW} ${svgH}`}
    78→          preserveAspectRatio="none"
    79→        >
    80→          <defs>
    81→            {/* Proximity fade gradient - chevrons near text are brighter */}
    82→            <linearGradient id={`fade-${direction}`} x1="0%" y1="0%" x2="100%" y2="0%">
    83→              {isLeft ? (
    84→                <>
    85→                  <stop offset="0%" stopColor={color} stopOpacity="0.06" />
    86→                  <stop offset="80%" stopColor={color} stopOpacity="0.45" />
    87→                  <stop offset="100%" stopColor={color} stopOpacity="0.6" />
    88→                </>
    89→              ) : (
    90→                <>
    91→                  <stop offset="0%" stopColor={color} stopOpacity="0.6" />
    92→                  <stop offset="20%" stopColor={color} stopOpacity="0.45" />
    93→                  <stop offset="100%" stopColor={color} stopOpacity="0.06" />
    94→                </>
    95→              )}
    96→            </linearGradient>
    97→
    98→            {/* Sweep gradient for animated glow - moves from edge toward text */}
    99→            <linearGradient id={`sweep-${direction}`} gradientUnits="userSpaceOnUse" x1="0" y1="0" x2={svgW} y2="0">
   100→              <stop stopColor="white" stopOpacity="0">
   101→                <animate
   102→                  attributeName="offset"
   103→                  values={isLeft ? '-0.25;-0.25;1.25' : '1.25;1.25;-0.25'}
   104→                  keyTimes="0;0.55;1"
   105→                  dur="2.2s"
   106→                  repeatCount="indefinite"
   107→                />
   108→              </stop>
   109→              <stop stopColor="white" stopOpacity="1">
   110→                <animate
   111→                  attributeName="offset"
   112→                  values={isLeft ? '-0.12;-0.12;1.38' : '1.38;1.38;-0.12'}
   113→                  keyTimes="0;0.55;1"
   114→                  dur="2.2s"
   115→                  repeatCount="indefinite"
   116→                />
   117→              </stop>
   118→              <stop stopColor="white" stopOpacity="0">
   119→                <animate
   120→                  attributeName="offset"
   121→                  values={isLeft ? '0;0;1.5' : '1.5;1.5;0'}
   122→                  keyTimes="0;0.55;1"
   123→                  dur="2.2s"
   124→                  repeatCount="indefinite"
   125→                />
   126→              </stop>
   127→            </linearGradient>
   128→
   129→            <mask id={`mask-${direction}`}>
   130→              <rect x="0" y="0" width={svgW} height={svgH} fill={`url(#sweep-${direction})`} />
   131→            </mask>
   132→          </defs>
   133→
   134→          {/* Base chevrons - always dimly visible */}
   135→          {Array.from({ length: count }).map((_, i) => {
   136→            const x = i * spacing
   137→            return (
   138→              <path
   139→                key={i}
   140→                d={isLeft
   141→                  ? `M${x + 4} 4 L${x + 10} 11 L${x + 4} 18`
   142→                  : `M${x + 10} 4 L${x + 4} 11 L${x + 10} 18`
   143→                }
   144→                fill="none"
   145→                stroke={`url(#fade-${direction})`}
   146→                strokeWidth="1.2"
   147→                strokeLinecap="round"
   148→                strokeLinejoin="round"
   149→              />
   150→            )
   151→          })}
   152→
   153→          {/* Glow chevrons - revealed by sweeping mask */}
   154→          <g mask={`url(#mask-${direction})`} style={{ filter: `drop-shadow(0 0 4px ${color})` }}>
   155→            {Array.from({ length: count }).map((_, i) => {
   156→              const x = i * spacing
   157→              return (
   158→                <path
   159→                  key={`g${i}`}
   160→                  d={isLeft
   161→                    ? `M${x + 4} 4 L${x + 10} 11 L${x + 4} 18`
   162→                    : `M${x + 10} 4 L${x + 4} 11 L${x + 10} 18`
   163→                  }
   164→                  fill="none"
   165→                  stroke={color}
   166→                  strokeWidth="2"
   167→                  strokeLinecap="round"
   168→                  strokeLinejoin="round"
   169→                />
   170→              )
   171→            })}
   172→          </g>
   173→        </svg>
   174→      </div>
   175→    )
   176→  }
   177→
   178→  return (
   179→    <div className="flex items-center w-full">
   180→      {/* Left arrows: flowing right → toward text */}
   181→      <ArrowStream direction="left" />
   182→
   183→      {/* N•M•C text */}
   184→      <div className="flex items-center justify-center gap-0.5 flex-shrink-0 px-1">
   185→        {parts.map((part, i) => {
   186→          const isBullet = part === '•'
   187→          return (
   188→            <motion.span
   189→              key={i}
   190→              className="relative inline-block leading-none"
   191→              style={{
   192→                fontSize: isBullet ? '1.5rem' : '3rem',
   193→                fontWeight: isBullet ? 400 : 900,
   194→                fontFamily: '"Outfit", system-ui, sans-serif',
   195→                color: isBullet ? `${color}60` : color,
   196→                textShadow: isBullet
   197→                  ? `0 0 6px ${color}30`
   198→                  : `0 0 10px ${color}80, 0 0 30px ${color}40, 0 0 60px ${color}20, 0 0 100px ${color}10`,
   199→                letterSpacing: isBullet ? '0' : '0.05em',
   200→              }}
   201→              initial={{ opacity: 0, y: 20 }}
   202→              animate={{ opacity: 1, y: 0 }}
   203→              transition={{
   204→                delay: 0.1 + i * 0.08,
   205→                duration: 0.5,
   206→                ease: [0.25, 0.46, 0.45, 0.94],
   207→              }}
   208→            >
   209→              {part}
   210→              {!isBullet && (
   211→                <motion.div
   212→                  className="absolute inset-0 pointer-events-none"
   213→                  style={{
   214→                    background: `linear-gradient(180deg, transparent 0%, ${color}30 50%, transparent 100%)`,
   215→                    backgroundSize: '100% 200%',
   216→                  }}
   217→                  animate={{
   218→                    backgroundPosition: ['0% 0%', '0% 200%'],
   219→                  }}
   220→                  transition={{
   221→                    duration: 2 + i * 0.3,
   222→                    repeat: Infinity,
   223→                    ease: 'linear',
   224→                    delay: i * 0.2,
   225→                  }}
   226→                />
   227→              )}
   228→            </motion.span>
   229→          )
   230→        })}
   231→      </div>
   232→
   233→      {/* Right arrows: flowing left ← toward text */}
   234→      <ArrowStream direction="right" />
   235→    </div>
   236→  )
   237→}
