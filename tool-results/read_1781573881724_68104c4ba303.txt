   401→function num(v: unknown): number {
   402→  if (v === null || v === undefined) return 0;
   403→  if (typeof v === 'number') return isFinite(v) ? v : 0;
   404→  const s = String(v).trim().replace(/\s/g, '').replace(/%/g, '');
   405→  if (!s) return 0;
   406→  const n = Number(s.replace(/[^0-9.\-]/g, ''));
   407→  return isFinite(n) ? n : 0;
   408→}
   409→function fmt(n: number): string { return new Intl.NumberFormat('vi-VN').format(n); }
   410→function fmtBig(v: number): string {
   411→  if (v >= 1e9) return (v/1e9).toFixed(2) + ' tỷ';
   412→  if (v >= 1e6) return (v/1e6).toFixed(0) + ' tr';
   413→  if (v >= 1e3) return (v/1e3).toFixed(0) + 'k';
   414→  return v.toFixed(0);
   415→}
   416→function fmtTyTrong(v: number): string {
   417→  if (!isFinite(v) || isNaN(v)) return '0%';
   418→  if (Math.abs(v % 1) < 0.001) return fmt(Math.round(v)) + '%';
   419→  return v.toLocaleString('vi-VN', { minimumFractionDigits: 0, maximumFractionDigits: 0 }) + '%';
   420→}
   421→function fmtShort(v: number): string {
   422→  if (v >= 1e9) return (v/1e9).toFixed(1) + ' tỷ';
   423→  if (v >= 1e6) return (v/1e6).toFixed(0) + ' tr';
   424→  if (v >= 1e3) return (v/1e3).toFixed(0) + 'k';
   425→  return v.toFixed(0);
   426→}

   428→  const p = Math.max(0, Math.min(100, pct || 0));

   484→function useAnimateValue(ref: React.RefObject<HTMLElement | null>, end: number, dur = 900) {
   485→  useEffect(() => {
   486→    const el = ref.current;
   487→    if (!el) return;
   488→    let st: number | null = null;
   489→    let raf: number;
   490→    const step = (ts: number) => {
   491→      if (!st) st = ts;
   492→      const p = Math.min((ts - st) / dur, 1);
   493→      const ease = 1 - Math.pow(1 - p, 3);
   494→      el.textContent = fmt(Math.floor(end * ease));
   495→      if (p < 1) raf = requestAnimationFrame(step);
   496→    };
   497→    raf = requestAnimationFrame(step);
   498→    return () => cancelAnimationFrame(raf);
   499→  }, [ref, end, dur]);
   500→}
   501→
   502→function useAnimatePct(ref: React.RefObject<HTMLElement | null>, target: number, dur = 900, dec = 0) {
   503→  useEffect(() => {
   504→    const el = ref.current;
   505→    if (!el) return;
   506→    let st: number | null = null;
   507→    let raf: number;
   508→    const step = (ts: number) => {
   509→      if (!st) st = ts;
   510→      const p = Math.min((ts - st) / dur, 1);
   511→      const ease = 1 - Math.pow(1 - p, 3);
   512→      el.textContent = (target * ease).toFixed(dec) + '%';
   513→      if (p < 1) raf = requestAnimationFrame(step);
   514→    };
   515→    raf = requestAnimationFrame(step);
   516→    return () => cancelAnimationFrame(raf);
   517→  }, [ref, target, dur, dec]);
   518→}
