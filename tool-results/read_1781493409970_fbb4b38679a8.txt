   376→  if (typeof v === 'number') return isFinite(v) ? v : 0;
   377→  const s = String(v).trim().replace(/\s/g, '').replace(/%/g, '');
   378→  if (!s) return 0;
   379→  const n = Number(s.replace(/[^0-9.\-]/g, ''));
   380→  return isFinite(n) ? n : 0;
   381→}
   382→function fmt(n: number): string { return new Intl.NumberFormat('vi-VN').format(n); }
   383→function fmtBig(v: number): string {
   384→  if (v >= 1e9) return (v/1e9).toFixed(2) + ' tỷ';
   385→  if (v >= 1e6) return (v/1e6).toFixed(0) + ' tr';
   386→  if (v >= 1e3) return (v/1e3).toFixed(0) + 'k';
   387→  return v.toFixed(0);
   388→}
   389→function fmtTyTrong(v: number): string {
   390→  if (!isFinite(v) || isNaN(v)) return '0%';
   391→  if (Math.abs(v % 1) < 0.001) return fmt(Math.round(v)) + '%';
   392→  return v.toLocaleString('vi-VN', { minimumFractionDigits: 0, maximumFractionDigits: 0 }) + '%';
   393→}
   394→function fmtShort(v: number): string {
   395→  if (v >= 1e9) return (v/1e9).toFixed(1) + ' tỷ';
   396→  if (v >= 1e6) return (v/1e6).toFixed(0) + ' tr';
   397→  if (v >= 1e3) return (v/1e3).toFixed(0) + 'k';
   398→  return v.toFixed(0);
   399→}
   400→function progressColor(pct: number): string {
   401→  const p = Math.max(0, Math.min(100, pct || 0));
   402→  const hue = 0 + (120 * (p / 100));
   403→  return `hsl(${hue}, 68%, 52%)`;
   404→}
   405→function glowCls(pct: number): string { return pct >= 100 ? ' glow-full' : ''; }
   406→function pctClass(pct: number): string { return pct >= 100 ? 'green' : pct >= 75 ? 'gold' : 'red'; }
   407→function normKey(v: string): string {
   408→  return String(v || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/Đ/g,'D').replace(/đ/g,'d').toUpperCase().replace(/[^A-Z0-9]/g, '');
   409→}
   410→function getMonthFromDate(d: string | null | undefined): string {
   411→  if (!d) return '';
   412→  try {
   413→    const dt = new Date(d);

   415→    return `${dt.getFullYear()}-${String(dt.getMonth()+1).padStart(2,'0')}`;
   416→  } catch { return ''; }
   417→}

   423→/* ================= ANIMATION HOOKS ================= */
   424→function useAnimateValue(ref: React.RefObject<HTMLElement | null>, end: number, dur = 900) {
   425→  useEffect(() => {
   426→    const el = ref.current;
   427→    if (!el) return;
   428→    let st: number | null = null;
   429→    let raf: number;
   430→    const step = (ts: number) => {
   431→      if (!st) st = ts;
   432→      const p = Math.min((ts - st) / dur, 1);
   433→      const ease = 1 - Math.pow(1 - p, 3);
   434→      el.textContent = fmt(Math.floor(end * ease));
   435→      if (p < 1) raf = requestAnimationFrame(step);
   436→    };
   437→    raf = requestAnimationFrame(step);
   438→    return () => cancelAnimationFrame(raf);
   439→  }, [ref, end, dur]);
   440→}
   441→
   442→function useAnimatePct(ref: React.RefObject<HTMLElement | null>, target: number, dur = 900, dec = 0) {
   443→  useEffect(() => {
   444→    const el = ref.current;
   445→    if (!el) return;
   446→    let st: number | null = null;
   447→    let raf: number;
   448→    const step = (ts: number) => {
   449→      if (!st) st = ts;
   450→      const p = Math.min((ts - st) / dur, 1);
   451→      const ease = 1 - Math.pow(1 - p, 3);
   452→      el.textContent = (target * ease).toFixed(dec) + '%';
   453→      if (p < 1) raf = requestAnimationFrame(step);
   454→    };
   455→    raf = requestAnimationFrame(step);
   456→    return () => cancelAnimationFrame(raf);
   457→  }, [ref, target, dur, dec]);
   458→}
