   114→function isTVVm(startDate: string | null, maxMonths: number = 12): boolean {
   115→  if (!startDate) return false;
   116→  const start = new Date(startDate);
   117→  const now = new Date();
   118→  const diffMonths = (now.getFullYear() - start.getFullYear()) * 12 + (now.getMonth() - start.getMonth());
   119→  return diffMonths <= maxMonths;
   120→}

   122→// TVV90: TVV có thời gian làm việc không quá N tháng
   123→function isTVV90Agent(contracts: Contract[], agentCode: string, maxMonths: number = 3, _minIP?: number): boolean {
   124→  const agentContract = contracts.find(c => c.agentCode === agentCode);
   125→  if (!agentContract) return false;
   126→  const startDate = agentContract.ngayBatDauLamViec || agentContract.startDate;
   127→  if (!startDate) return false;
   128→  const start = new Date(startDate);
   129→  const now = new Date();
   130→  const diffMonths = (now.getFullYear() - start.getFullYear()) * 12 + (now.getMonth() - start.getMonth());
   131→  return diffMonths <= maxMonths;
   132→}

   134→// Helper: calculate lượt for a group of contracts based on tinhLuot3tr
   135→// Đếm SỐ DÒNG hợp đồng có tinhLuot3tr >= threshold (không phải unique TVV)
   136→function calculateLuot(contracts: Contract[], luotThreshold: number, conditionType: ConditionType, tvv90MaxMonths?: number, tvv90MinIP?: number): number {
   137→  let count = 0;
   138→  for (const c of contracts) {
   139→    // Apply TVVm filter if condition is tvvm mode
   140→    if (isTVVmMode(conditionType)) {
   141→      if (!isTVVm(c.ngayBatDauLamViec || c.startDate)) continue;
   142→    }
   143→    // Apply TVV90 filter if condition is tvv90 mode
   144→    if (conditionType === 'activity_round_tvv90') {
   145→      if (!isTVV90Agent(contracts, c.agentCode, tvv90MaxMonths, tvv90MinIP)) continue;
   146→    }
   147→    // Đếm số dòng hợp đồng có tinhLuot3tr >= threshold
   148→    if (c.tinhLuot3tr >= luotThreshold) {
   149→      count++;
   150→    }
   151→  }
   152→  return count;
   153→}

   168→function formatBonus(tier: BonusTier, fyp?: number, rounds?: number): string {
   169→  if (tier.bonusType === 'gift' && tier.bonusText) return tier.bonusText;
   170→  if (tier.bonusType === 'percent' && tier.bonusPercent > 0) {
   171→    const calculated = fyp ? tier.bonusPercent / 100 * fyp : 0;
   172→    return `${tier.bonusPercent}% IP${fyp ? ` = ${formatCurrency(calculated)}` : ''}`;
   173→  }
   174→  if (tier.bonusType === 'percent_fyc' && tier.bonusPercent > 0) {
   175→    const fyc = fyp ? fyp * 0.25 : 0;
   176→    const calculated = fyc ? tier.bonusPercent / 100 * fyc : 0;
   177→    return `${tier.bonusPercent}% FYC${fyp ? ` = ${formatCurrency(calculated)}` : ''}`;
   178→  }
   179→  if (tier.bonusType === 'money_per_round') {
   180→    const calculated = rounds ? tier.bonusAmount * rounds : 0;
   181→    return rounds ? `${formatCurrency(tier.bonusAmount)}/lượt × ${rounds} = ${formatCurrency(calculated)}` : `${formatCurrency(tier.bonusAmount)}/lượt`;
   182→  }
   183→  return formatCurrency(tier.bonusAmount);
   184→}

   186→// Format rate label for percent-based bonus types
   187→function formatRate(tier: BonusTier): string {
   188→  if (tier.bonusType === 'percent') return `${tier.bonusPercent}%`;
   189→  if (tier.bonusType === 'percent_fyc') return `${tier.bonusPercent}%`;

   191→}

   193→// Format bonus amount only (no formula) for display in table
   194→function formatBonusAmount(tier: BonusTier, fyp?: number, rounds?: number): string {
   195→  if (tier.bonusType === 'gift' && tier.bonusText) return tier.bonusText;
   196→  const amount = computeBonusFromTier(tier, fyp || 0, rounds);
   197→  return formatCurrency(amount);
   198→}

   205→function computeBonusFromTier(tier: BonusTier, fyp: number, rounds?: number): number {
   206→  if (tier.bonusType === 'percent') return tier.bonusPercent / 100 * fyp;
   207→  if (tier.bonusType === 'percent_fyc') return tier.bonusPercent / 100 * (fyp * 0.25);
   208→  if (tier.bonusType === 'money_per_round') return tier.bonusAmount * (rounds || 0);
   209→  return tier.bonusAmount;
   210→}
