   119→function isTVVm(startDate: string | null, maxMonths: number = 12): boolean {
   120→  if (!startDate) return false;
   121→  const start = new Date(startDate);
   122→  const now = new Date();
   123→  const diffMonths = (now.getFullYear() - start.getFullYear()) * 12 + (now.getMonth() - start.getMonth());
   124→  return diffMonths <= maxMonths;
   125→}

   127→// TVV90: TVV có thời gian làm việc không quá N tháng
   128→function isTVV90Agent(contracts: Contract[], agentCode: string, maxMonths: number = 3, _minIP?: number): boolean {
   129→  const agentContract = contracts.find(c => c.agentCode === agentCode);
   130→  if (!agentContract) return false;
   131→  const startDate = agentContract.ngayBatDauLamViec || agentContract.startDate;
   132→  if (!startDate) return false;
   133→  const start = new Date(startDate);
   134→  const now = new Date();
   135→  const diffMonths = (now.getFullYear() - start.getFullYear()) * 12 + (now.getMonth() - start.getMonth());
   136→  return diffMonths <= maxMonths;
   137→}

   139→// Helper: calculate lượt for a group of contracts based on tinhLuot3tr
   140→// Đếm SỐ DÒNG hợp đồng có tinhLuot3tr >= threshold (không phải unique TVV)
   141→function calculateLuot(contracts: Contract[], luotThreshold: number, conditionType: ConditionType, tvv90MaxMonths?: number, tvv90MinIP?: number): number {
   142→  let count = 0;
   143→  for (const c of contracts) {
   144→    // Apply TVVm filter if condition is tvvm mode
   145→    if (isTVVmMode(conditionType)) {
   146→      if (!isTVVm(c.ngayBatDauLamViec || c.startDate)) continue;
   147→    }
   148→    // Apply TVV90 filter if condition is tvv90 mode
   149→    if (conditionType === 'activity_round_tvv90') {
   150→      if (!isTVV90Agent(contracts, c.agentCode, tvv90MaxMonths, tvv90MinIP)) continue;
   151→    }
   152→    // Đếm số dòng hợp đồng có tinhLuot3tr >= threshold
   153→    if (c.tinhLuot3tr >= luotThreshold) {
   154→      count++;
   155→    }
   156→  }
   157→  return count;
   158→}

   173→function formatBonus(tier: BonusTier, fyp?: number, rounds?: number): string {
   174→  if (tier.bonusType === 'gift' && tier.bonusText) return tier.bonusText;
   175→  if (tier.bonusType === 'percent' && tier.bonusPercent > 0) {
   176→    const calculated = fyp ? tier.bonusPercent / 100 * fyp : 0;
   177→    return `${tier.bonusPercent}% IP${fyp ? ` = ${formatCurrency(calculated)}` : ''}`;
   178→  }
   179→  if (tier.bonusType === 'percent_fyc' && tier.bonusPercent > 0) {
   180→    const fyc = fyp ? fyp * 0.25 : 0;
   181→    const calculated = fyc ? tier.bonusPercent / 100 * fyc : 0;
   182→    return `${tier.bonusPercent}% FYC${fyp ? ` = ${formatCurrency(calculated)}` : ''}`;
   183→  }
   184→  if (tier.bonusType === 'money_per_round') {
   185→    const calculated = rounds ? tier.bonusAmount * rounds : 0;
   186→    return rounds ? `${formatCurrency(tier.bonusAmount)}/lượt × ${rounds} = ${formatCurrency(calculated)}` : `${formatCurrency(tier.bonusAmount)}/lượt`;
   187→  }
   188→  return formatCurrency(tier.bonusAmount);
   189→}

   192→function formatRate(tier: BonusTier): string {
   193→  if (tier.bonusType === 'percent') return `${tier.bonusPercent}%`;

   198→// Format bonus amount only (no formula) for display in table
   199→function formatBonusAmount(tier: BonusTier, fyp?: number, rounds?: number): string {
   200→  if (tier.bonusType === 'gift' && tier.bonusText) return tier.bonusText;
   201→  const amount = computeBonusFromTier(tier, fyp || 0, rounds);
   202→  return formatCurrency(amount);
   203→}

   210→function computeBonusFromTier(tier: BonusTier, fyp: number, rounds?: number): number {
   211→  if (tier.bonusType === 'percent') return tier.bonusPercent / 100 * fyp;
   212→  if (tier.bonusType === 'percent_fyc') return tier.bonusPercent / 100 * (fyp * 0.25);
   213→  if (tier.bonusType === 'money_per_round') return tier.bonusAmount * (rounds || 0);
   214→  return tier.bonusAmount;
   215→}
