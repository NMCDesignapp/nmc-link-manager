   451→function num(v: unknown): number {
   452→  if (v === null || v === undefined) return 0;
   453→  if (typeof v === 'number') return isFinite(v) ? v : 0;
   454→  const s = String(v).trim().replace(/\s/g, '').replace(/%/g, '');
   455→  if (!s) return 0;
   456→  const n = Number(s.replace(/[^0-9.\-]/g, ''));
   457→  return isFinite(n) ? n : 0;
   458→}

   534→function useAnimateValue(ref: React.RefObject<HTMLElement | null>, end: number, dur = 900) {
   535→  useEffect(() => {
   536→    const el = ref.current;
   537→    if (!el) return;
   538→    let st: number | null = null;
   539→    let raf: number;
   540→    const step = (ts: number) => {
   541→      if (!st) st = ts;
   542→      const p = Math.min((ts - st) / dur, 1);
   543→      const ease = 1 - Math.pow(1 - p, 3);
   544→      el.textContent = fmt(Math.floor(end * ease));
   545→      if (p < 1) raf = requestAnimationFrame(step);
   546→    };
   547→    raf = requestAnimationFrame(step);
   548→    return () => cancelAnimationFrame(raf);
   549→  }, [ref, end, dur]);
   550→}

   552→function useAnimatePct(ref: React.RefObject<HTMLElement | null>, target: number, dur = 900, dec = 0) {
   553→  useEffect(() => {
   554→    const el = ref.current;
   555→    if (!el) return;
   556→    let st: number | null = null;
   557→    let raf: number;
   558→    const step = (ts: number) => {
   559→      if (!st) st = ts;
   560→      const p = Math.min((ts - st) / dur, 1);
   561→      const ease = 1 - Math.pow(1 - p, 3);
   562→      el.textContent = (target * ease).toFixed(dec) + '%';
   563→      if (p < 1) raf = requestAnimationFrame(step);
   564→    };
   565→    raf = requestAnimationFrame(step);
   566→    return () => cancelAnimationFrame(raf);
   567→  }, [ref, target, dur, dec]);
   568→}
